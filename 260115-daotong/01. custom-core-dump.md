```python
# Patch view_as_complex to use manual complex multiplication for ONNX compatibility
def reshape_for_broadcast_onnx(freqs_cis, x):
    ndim = x.ndim
    assert freqs_cis.shape == (x.shape[-2], x.shape[-1])
    shape = [d if i >= ndim - 2 else 1 for i, d in enumerate(x.shape)]
    return freqs_cis.view(*shape)

def apply_rotary_enc_onnx_compatible(xq, xk, freqs_cis, repeat_freqs_k=False):
    # freqs_cis is already converted to [seq_len, d//2, 2] format
    freqs_real_base = freqs_cis[..., 0] # [seq_len, d//2]
    freqs_imag_base = freqs_cis[..., 1] # [seq_len, d//2]

    xq_reshaped = xq.float().reshape(*xq.shape[:-1], -1, 2)
    xq_real, xq_imag = xq_reshaped[..., 0], xq_reshaped[..., 1]

    # Broadcast: [seq_len, d//2] -> [1, ..., 1, seq_len, d//2]
    ndim = xq_real.ndim
    shape = [d if i >= ndim - 2 else 1 for i, d in enumerate(xq_real.shape)]
    freqs_real = freqs_real_base.view(*shape)
    freqs_imag = freqs_imag_base.view(*shape)

    xq_out_real = xq_real * freqs_real - xq_imag * freqs_imag
    xq_out_imag = xq_real * freqs_imag + xq_imag * freqs_real
    xq_out = torch.stack([xq_out_real, xq_out_imag], dim=-1).flatten(-2)

    if xk.shape[-2] == 0:
        return xq_out.type_as(xq), xk

    xk_reshaped = xk.float().reshape(*xk.shape[:-1], -1, 2)
    xk_real, xk_imag = xk_reshaped[..., 0], xk_reshaped[..., 1]
    
    if repeat_freqs_k:
        r = xk_real.shape[-2] // xq_real.shape[-2]
        freqs_real_k_base = freqs_real_base.repeat(r, 1)
        freqs_imag_k_base = freqs_imag_base.repeat(r, 1)
    else:
        freqs_real_k_base = freqs_real_base
        freqs_imag_k_base = freqs_imag_base
    ndim_k = xk_real.ndim
    shape_k = [d if i >= ndim_k - 2 else 1 for i, d in enumerate(xk_real.shape)]
    freqs_real_k = freqs_real_k_base.view(*shape_k)
    freqs_imag_k = freqs_imag_k_base.view(*shape_k)

    xk_out_real = xk_real * freqs_real_k - xk_imag * freqs_imag_k
    xk_out_imag = xk_real * freqs_imag_k + xk_imag * freqs_real_k
    xk_out = torch.stack([xk_out_real, xk_out_imag], dim=-1).flatten(-2)

    return xq_out.type_as(xq), xk_out.type_as(xk)

class Sam3ONNXWrapper(torch.nn.Module):
    def __init__(self, sam3_model, device="cpu"):
        super().__init__()
        self.model = sam3_model
        self.device = torch.device(device)

    def forward(self, img_batch, language_features, language_mask):
        from sam3.model.data_misc import FindStage, BatchedFindTarget
        from sam3.model.geometry_encoders import Prompt

        device = self.device
        backbone_out = {"img_batch_all_stages": img_batch}
        backbone_out.update(self.model.backbone.forward_image(img_batch))
        backbone_out["language_features"] = language_features
        backbone_out["language_mask"] = language_mask

        # 使用dummy box来匹配geometric_prompt
        dummy_box_tensor = torch.tensor([[0.0, 0.0, 0.001, 0.001]], device=device) # [1, 4]
        find_input = FindStage(
            img_ids=torch.tensor([0], dtype=torch.long, device=device),
            text_ids=torch.tensor([0], dtype=torch.long, device=device),
            input_boxes=[dummy_box_tensor], # 使用dummy box
            input_boxes_mask=[torch.ones(1, dtype=torch.bool, device=device)], # 全部mask掉
            input_boxes_label=[torch.zeros(1, dtype=torch.long, device=device)], # 负样本标签
            input_points=[torch.empty(0, 257, device=device)],
            input_points_mask=[torch.empty(0, device=device)],
            object_ids=[],
        )

        find_target = BatchedFindTarget(
            num_boxes=[], boxes=[], boxes_padded=[], is_exhaustive=[],
            segments=[], semantic_segments=[], is_valid_segment=[],
            repeated_boxes=[], object_ids=[], object_ids_padded=[],
        )

        # 使用一个非空的dummy box来避免roi_align的形状推断问题
        # 这个box很小且位于图像角落，不会影响模型输出（会被mask过滤掉）
        # 形状: [1, 1, 4] = [n_boxes, bs, 4]
        dummy_box = torch.tensor([[[0.0, 0.0, 0.001, 0.001]]], device=device) # 很小的box在左上角
        geometric_prompt = Prompt(
            box_embeddings=dummy_box, # [1, 1, 4]
            box_mask=torch.ones(1, 1, device=device, dtype=torch.bool), # 全部mask掉，不影响输出
            box_labels=torch.zeros(1, 1, device=device, dtype=torch.long), #负样本标签
        )

        out = self.model.forward_grounding(
            backbone_out=backbone_out,
            find_input=find_input,
            find_target=find_target,
            geometric_prompt=geometric_prompt,
        )
        return out["pred_masks"], out["pred_boxes"], out["pred_logits"]

wrapper = Sam3ONNXWrapper(model, device=device).to(device).eval()

print("5. Testing wrapper forward pass...")
clear_decoder_cache(model)
with torch.no_grad():
    test_output = wrapper(batch.img_batch, language_features, language_mask)
```

```python
def pt_inference_npu(batch, model):
    print("\n" + "="*60)
    print("Step 1: PyTorch Inference (NPU)")
    print("="*60)

    clear_decoder_cache(model)
    batch_npu = copy_data_to_device(batch, torch.device("npu:4"), non_blocking=True)   # 这里，硬编码使用 npuid 为 4 的设备，

    with torch.no_grad():
        output = model(batch_npu)

    postprocessor = PostProcessImage(
        max_dets_per_img=-1,
        iou_type="segm",
        use_original_sizes_box=True,
        use_original_sizes_mask=True,
        convert_mask_to_rle=False,
        detection_threshold=0.5,
        to_cpu=False,
        always_interpolate_masks_on_gpu=False,
    )

    processed_results = postprocessor.process_results(output, batch_npu.find_metadatas)
    result = processed_results[1]

    print(f"PT Inference Success: {len(result['masks'])} masks detected")
    return result, output

def onnx_inference_cpu(batch):

    print("Step 2: ONNX Inference (CPU)")
    print("="*60)

    model_cpu.eval()

        text_outputs = model_cpu.backbone.forward_text(batch_cpu.find_text_batch, device="cpu")

    outputs = session.run(None, {
        "img_batch": batch_cpu.img_batch.cpu().numpy().astype(np.float32),
        "language_features": text_outputs["language_features"].cpu().numpy().astype(np.float32),
        "language_mask": text_outputs["language_mask"].cpu().numpy().astype(np.bool_),
    })

    print(f"ONNX Inference Success: masks={outputs[0].shape}, boxes={outputs[1].shape}, logits={outputs[2].shape}")
    return torch.from_numpy(outputs[0]), torch.from_numpy(outputs[1]), torch.from_numpy(outputs[2])

def compare_results(pt_output, onnx_outputs):
    print("\n" + "="*60)
    print("Step 3: Compare Results (Cosine Similarity)")
    print("="*60)

    output_list = pt_output.output if hasattr(pt_output, 'output') else pt_output
    pt_output_dict = output_list[0][0]
    pred_masks_pt = pt_output_dict["pred_masks"].cpu()
    pred_boxes_pt = pt_output_dict["pred_boxes"].cpu()
    pred_logits_pt = pt_output_dict["pred_logits"].cpu()

    pred_masks_onnx, pred_boxes_onnx, pred_logits_onnx = onnx_outputs

    # Align shapes for comparison
    if pred_masks_pt.ndim == 4 and pred_masks_pt.shape[1] == 1:
        pred_masks_pt = pred_masks_pt.permute(1, 0, 2, 3)
    if pred_masks_pt.shape[1] < pred_masks_onnx.shape[1]:
        pad_size = pred_masks_onnx.shape[1] - pred_masks_pt.shape[1]
        pred_masks_pt = torch.cat([pred_masks_pt, torch.zeros(1, pad_size, *pred_masks_pt.shape[2:])], dim=1)
    elif pred_masks_pt.shape[1] > pred_masks_onnx.shape[1]:
        pred_masks_pt = pred_masks_pt[:, :pred_masks_onnx.shape[1]]

    if pred_boxes_pt.ndim == 2:
        pred_boxes_pt = pred_boxes_pt.unsqueeze(0)
    if pred_logits_pt.ndim == 2:
        pred_logits_pt = pred_logits_pt.unsqueeze(0)

    if pred_boxes_pt.shape[1] < pred_boxes_onnx.shape[1]:
        pad_size = pred_boxes_onnx.shape[1] - pred_boxes_pt.shape[1]
        pred_boxes_pt = torch.cat([pred_boxes_pt, torch.zeros(1, pad_size, 4)], dim=1)
        pred_logits_pt = torch.cat([pred_logits_pt, torch.zeros(1, pad_size, 1)], dim=1)
    elif pred_boxes_pt.shape[1] > pred_boxes_onnx.shape[1]:
        pred_boxes_pt = pred_boxes_pt[:, :pred_boxes_onnx.shape[1]]
        pred_logits_pt = pred_logits_pt[:, :pred_logits_onnx.shape[1]]

    if pred_masks_pt.shape[-2:] != pred_masks_onnx.shape[-2:]:
        pred_masks_pt = torch.nn.functional.interpolate(pred_masks_pt, size=pred_masks_onnx.shape[-2:], mode='bilinear', align_corners=False)

    sim_masks = cosine_similarity_tensor(pred_masks_pt, pred_masks_onnx)
    sim_boxes = cosine_similarity_tensor(pred_boxes_pt, pred_boxes_onnx)
    sim_logits = cosine_similarity_tensor(pred_logits_pt, pred_logits_onnx)
    avg_sim = (sim_masks + sim_boxes + sim_logits) / 3

    print(f"\nCosine Similarity Results:")
    print(f" pred_masks: {sim_masks:.6f} ({sim_masks*100:.2f}%)")
    print(f" pred_boxes: {sim_boxes:.6f} ({sim_boxes*100:.2f}%)")
    print(f" pred_logits: {sim_logits:.6f} ({sim_logits*100:.2f}%)")
    print(f" Average: {avg_sim:.6f} ({avg_sim*100:.2f}%)")
    print(" Similarity >= 99%" if avg_sim >= 0.99 else f" Similarity < 99% (current: {avg_sim*100:.2f}%)")

    return sim_masks, sim_boxes, sim_logits

def convert_onnx_outputs_to_results(onnx_outputs, original_image, threshold=0.5):
    pred_masks, pred_boxes, pred_logits = onnx_outputs
    scores = torch.sigmoid(pred_logits.squeeze(-1))
    valid_indices = scores[0] > threshold

    if valid_indices.sum() == 0:
        return {"masks": [], "boxes": [], "scores": []}

    masks_list, boxes_list, scores_list = [], [], []
    w, h = original_image.size

    for i in range(pred_masks.shape[1]):
        if valid_indices[i]:
            mask = pred_masks[0, i] if pred_masks.ndim == 4 else pred_masks[0, i, 0]
            if mask.shape != (h, w):
                mask = torch.nn.functional.interpolate(mask.unsqueeze(0).unsqueeze(0), size=(h, w), mode='bilinear', align_corners=False).squeeze()
            masks_list.append(((mask > 0.5).float()).cpu())

            box_cxcywh = pred_boxes[0, i]
            box_xyxy_norm = box_cxcywh_to_xyxy(box_cxcywh.unsqueeze(0)).squeeze(0)
            boxes_list.append((box_xyxy_norm * torch.tensor([w, h, w, h])).cpu())
            scores_list.append(scores[0, i].cpu())

    return {"masks": masks_list, "boxes": boxes_list, "scores": scores_list}

def process_mask(mask, target_h, target_w):
    """统一处理mask：调整大小和类型转换"""
    mask = mask.squeeze(0).cpu()
    if mask.dtype == torch.bool or mask.dtype == torch.uint8:
        mask = mask.float()
    if mask.shape != (target_h, target_w):
        mask = torch.nn.functional.interpolate(mask.unsqueeze(0).unsqueeze(0), size=(target_h, target_w), mode='bilinear', align_corners=False).squeeze()
    return mask
	
def clamp_box(box, w, h):
    """裁剪box到图像边界"""
    box = box.cpu().clone()
    box[0] = torch.clamp(box[0], 0, w)
    box[1] = torch.clamp(box[1], 0, h)
    box[2] = torch.clamp(box[2], 0, w)
    box[3] = torch.clamp(box[3], 0, h)
    return box
```

```python
import onnx
from onnx import TensorProto
import numpy as np
print('📥 加载原始ONNX模型...')
model = onnx.load('sam3.onnx', load_external_data=False)
graph = model.graph
print('🔧 深度修复Greater操作的数据类型不匹配...')
new_nodes = []
cast_count = 0
for node in graph.node:
    if node.op_type == 'Greater':
        print(f' 🎯 找到Greater节点: {node.name}')
        print(f' 输入1: {node.input[0]}')
        print(f' 输入2: {node.input[1]}')
        # 创建两个Cast节点，确保两个输入都是int32
        cast_node1_name = f'cast_greater_input1_{cast_count}'
        cast_output1_name = f'{node.input[0]}_cast_int32'
        cast_node2_name = f'cast_greater_input2_{cast_count}'
        cast_output2_name = f'{node.input[1]}_cast_int32'
        # Cast第一个输入到int32
        cast_node1 = onnx.helper.make_node(
            'Cast',
            inputs=[node.input[0]],
            outputs=[cast_output1_name],
            to=TensorProto.INT32,
            name=cast_node1_name
        )
        new_nodes.append(cast_node1)
        # Cast第二个输入到int32
        cast_node2 = onnx.helper.make_node(
            'Cast',
            inputs=[node.input[1]],
            outputs=[cast_output2_name],
            to=TensorProto.INT32,
            name=cast_node2_name
        )
        new_nodes.append(cast_node2)
        # 修改Greater节点使用cast后的输入
        fixed_greater = onnx.helper.make_node(
            'Greater',
            inputs=[cast_output1_name, cast_output2_name],
            outputs=node.output,
            name=node.name
        )
        new_nodes.append(fixed_greater)
        cast_count += 1
        print(f' ✅ 已添加两个Cast节点: {cast_node1_name}, {cast_node2_name}')
    else:
        new_nodes.append(node)
# 更新图
graph.ClearField('node')
graph.node.extend(new_nodes)
print('💾 保存彻底修复后的模型...')
onnx.save(model, 'customer_sam3_fixed.onnx')
print('✅ 已保存为 customer_sam3_fixed.onnx')
print(f'📊 总共修复了 {cast_count} 个Greater节点')
# 验证修复
print('🔍 验证修复结果...')
try:
    import onnxruntime as ort
    session = ort.InferenceSession('customer_sam3_fixed.onnx', providers= ['CPUExecutionProvider'])
    print('✅ ONNX模型验证成功！')
except Exception as e:
    print(f'❌ ONNX模型验证失败: {e}')
```

## 添加算子节点
导出 sam3_static.onnx 权重后，避免转 om 过程中出现 Greater 算子出现 INT32 和 INT64 数据类型冲突，

在路径 /data/fuyuxin/onnx_weights 通过执行脚本 onnx-weights-external.py 实现，代码如下，
```python
import onnx
from onnx import TensorProto
import numpy as np

def fix_onnx_model():
   print('📥 加载原始ONNX模型...')
   model = onnx.load('sam3_static.onnx', load_external_data=False)
   graph = model.graph
   
   print('🔧 深度修复Greater操作的数据类型不匹配...')
   new_nodes = []
   cast_count = 0
   
   for node in graph.node:
       if node.op_type == 'Greater':
           print(f' 🎯 找到Greater节点: {node.name}')
           print(f'   输入1: {node.input[0]}')
           print(f'   输入2: {node.input[1]}')
           
           # 创建两个Cast节点，确保两个输入都是int32
           cast_node1_name = f'cast_greater_input1_{cast_count}'
           cast_output1_name = f'{node.input[0]}_cast_int32'
           cast_node2_name = f'cast_greater_input2_{cast_count}'
           cast_output2_name = f'{node.input[1]}_cast_int32'
           
           # Cast第一个输入到int32
           cast_node1 = onnx.helper.make_node(
               'Cast',
               inputs=[node.input[0]],
               outputs=[cast_output1_name],
               to=TensorProto.INT32,
               name=cast_node1_name
           )
           new_nodes.append(cast_node1)
           
           # Cast第二个输入到int32
           cast_node2 = onnx.helper.make_node(
               'Cast',
               inputs=[node.input[1]],
               outputs=[cast_output2_name],
               to=TensorProto.INT32,
               name=cast_node2_name
           )
           new_nodes.append(cast_node2)
           
           # 修改Greater节点使用cast后的输入
           fixed_greater = onnx.helper.make_node(
               'Greater',
               inputs=[cast_output1_name, cast_output2_name],
               outputs=node.output,
               name=node.name
           )
           new_nodes.append(fixed_greater)
           
           cast_count += 1
           print(f' ✅ 已添加两个Cast节点: {cast_node1_name}, {cast_node2_name}')
       else:
           new_nodes.append(node)
   
   # 更新图
   graph.ClearField('node')
   graph.node.extend(new_nodes)
   
   print('💾 保存彻底修复后的模型...')
   onnx.save(model, 'sam3_static_fixed.onnx')
   print('✅ 已保存为 sam3_static_fixed.onnx')
   print(f'📊 总共修复了 {cast_count} 个Greater节点')
   
   # 验证修复
   print('🔍 验证修复结果...')
   try:
       import onnxruntime as ort
       session = ort.InferenceSession(
           'sam3_static_fixed.onnx',
           providers=['CPUExecutionProvider']
       )
       print('✅ ONNX模型验证成功！')
   except Exception as e:
       print(f'❌ ONNX模型验证失败: {e}')

if __name__ == '__main__':
   fix_onnx_model()
```

```python
            result = {"masks": [], "boxes": [], "scores": []}
        else:
            masks_list = []
            boxes_list = []
            scores_list = []
            for i in range(pred_masks.shape[1]):
                if valid_indices[i]:
                    # 修正反斜杠换行问题
                    mask = pred_masks[0, i] if pred_masks.ndim == 4 else pred_masks[0, i, 0]
                    masks_list.append(((mask > 0.5).float()).cpu())
                    boxes_list.append(pred_boxes[0, i].cpu())
                    scores_list.append(scores[0, i].cpu())
            result = {"masks": masks_list, "boxes": boxes_list, "scores": scores_list}
        
        print(f"OM推理成功: {len(result['masks'])}个掩码")
        return result
        
    except Exception as e:
        print(f"OM推理失败: {e}")
        print(f"错误详情: {traceback.format_exc()}")
        # 返回空结果以保持一致性
        return {"masks": [], "boxes": [], "scores": []}
```
